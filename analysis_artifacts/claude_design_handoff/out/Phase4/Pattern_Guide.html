코드 스니펫 — EmptyState 컴포넌트 
// components/ui/EmptyState.tsx
'use client';
import type { ReactNode } from 'react';

type EmptyVariant = 'default' | 'search' | 'permission' | 'error';

// error 변형은 role="alert" — 스크린 리더 즉시 읽음
const ROLE_MAP: Record<EmptyVariant, string> = {
  default: 'region', search: 'region', permission: 'region', error: 'alert',
};

export function EmptyState({
  variant, icon, title, description, action, className,
}: {
  variant: EmptyVariant; icon: string; title: string;
  description?: string; action?: ReactNode; className?: string;
}) {
  return (
    <div role={ROLE_MAP[variant]} aria-label={title}
      className={`empty-state ${className ?? ''}`}>
      <span aria-hidden="true" className="empty-icon">{icon}</span>
      <p className="empty-title">{title}</p>
      {description && <p className="empty-desc">{description}</p>}
      {action}
    </div>
  );
}
// 사용 예: <EmptyState variant="error" icon="⚠️" title="로드 실패"
//   action={<button className="btn btn-danger" onClick={retry}>다시 시도</button>} />

♿ 접근성 체크 (JM6)

  • 에러 변형은 role="alert" — 스크린 리더가 즉시 읽음
  • 장식 아이콘에 aria-hidden="true" 필수
  • 액션 버튼은 <button> 태그, 설명적인 텍스트 포함
  • 빈 상태 컨테이너에 aria-label로 문맥 제공
  • 모바일 액션 버튼 터치 타깃 ≥ 44px (height: var(--touch-target))

// components/ui/Loading.tsx
'use client';
import { useId } from 'react';

/** 인라인 스피너 — role="status" aria-label 필수 */
export function Spinner({ label = '로딩 중' }: { label?: string }) {
  return <span role="status" aria-label={label} className="spinner" />;
}

/** 스켈레톤 블록 — aria-hidden, 부모에 aria-busy="true" */
export function Skeleton({ width, height = 14, className }: {
  width?: string; height?: number; className?: string;
}) {
  return (
    <div aria-hidden="true" className={`skeleton ${className ?? ''}`}
      style={{ width: width ?? '100%', height }} />
  );
}

/** 진행률 표시 — role="progressbar" + aria-valuenow/min/max */
export function ProgressBar({ value, label }: { value: number; label: string }) {
  const id = useId();
  return (
    <div>
      <div id={id} className="text-sm text-muted">{label} {value}%</div>
      <div role="progressbar" aria-valuenow={value}
        aria-valuemin={0} aria-valuemax={100} aria-labelledby={id}
        className="progress-track">
        <div className="progress-fill" style={{ width: `${value}%` }} />
      </div>
    </div>
  );
}

/** 스켈레톤 래퍼 */
export function SkeletonCard({ children }: { children: React.ReactNode }) {
  return <div aria-busy="true" aria-label="콘텐츠 로딩 중">{children}</div>;
}

♿ 접근성 체크 (JM6)

  • 스피너: role="status" + aria-label="로딩 중"
  • 스켈레톤 컨테이너: aria-busy="true", 개별 블록은 aria-hidden="true"
  • 프로그레스: role="progressbar" + aria-valuenow/min/max 필수
  • prefers-reduced-motion: 애니메이션 비활성화 CSS 적용
  • 로딩 중 버튼에 disabled + aria-disabled="true"

코드 스니펫 — ConfirmModal (포커스 트랩 포함) 
// components/ui/ConfirmModal.tsx
'use client';
import { useEffect, useRef, useState } from 'react';

type ConfirmVariant = 'default' | 'destructive' | 'irreversible';

export function ConfirmModal({
  variant, title, description, confirmLabel,
  onConfirm, onCancel, confirmText = 'DELETE', isOpen,
}: {
  variant: ConfirmVariant; title: string; description: string;
  confirmLabel: string; onConfirm: () => void; onCancel: () => void;
  confirmText?: string; isOpen: boolean;
}) {
  const dialogRef = useRef<HTMLDialogElement>(null);
  const [typed, setTyped] = useState('');
  const isDestructive = variant !== 'default';
  const canConfirm = variant === 'irreversible' ? typed === confirmText : true;

  // 포커스 트랩 + ESC 닫기 — <dialog> 네이티브가 기본 지원
  useEffect(() => {
    if (!isOpen) return;
    dialogRef.current?.showModal?.();
    const onKey = (e: KeyboardEvent) => { if (e.key === 'Escape') onCancel(); };
    document.addEventListener('keydown', onKey);
    return () => { document.removeEventListener('keydown', onKey); dialogRef.current?.close?.(); };
  }, [isOpen, onCancel]);

  if (!isOpen) return null;
  return (
    <dialog ref={dialogRef}
      role={isDestructive ? 'alertdialog' : 'dialog'}
      aria-modal="true" aria-labelledby="cm-title" aria-describedby="cm-desc"
      className="modal-box"
    >
      <h3 id="cm-title">{title}</h3>
      <p id="cm-desc">{description}</p>
      {variant === 'irreversible' && (
        <div className="form-group">
          <label htmlFor="irr-input">확인을 위해 <strong>{confirmText}</strong> 입력</label>
          <input id="irr-input" type="text" autoComplete="off"
            value={typed} onChange={e => setTyped(e.target.value)} />
        </div>
      )}
      <div className="modal-actions">
        <button className="btn btn-ghost" onClick={onCancel}>취소</button>
        <button className={`btn ${isDestructive ? 'btn-danger' : 'btn-primary'}`}
          onClick={onConfirm} disabled={!canConfirm} aria-disabled={!canConfirm}>
          {confirmLabel}
        </button>
      </div>
    </dialog>
  );
}

♿ 접근성 체크 (JM6)

  • 일반 모달: role="dialog" aria-modal="true"
  • 파괴적/비가역: role="alertdialog" — 스크린 리더가 즉시 읽음
  • 포커스 트랩: 모달 내 포커스가 순환, 배경 클릭·Tab 탈출 차단
  • ESC 키로 항상 닫을 수 있어야 함
  • 닫힌 후 트리거 버튼으로 포커스 복귀
  • <dialog> 네이티브 요소 사용 권장 (기본 포커스 트랩·ESC 지원)

// hooks/useToast.ts
'use client';
import { useState, useCallback, useId } from 'react';

type ToastVariant = 'info' | 'success' | 'warning' | 'error';
interface Toast { id: string; variant: ToastVariant; title: string; message?: string; }

export function useToast() {
  const [toasts, setToasts] = useState<Toast[]>([]);
  const baseId = useId();

  const dismiss = useCallback((id: string) =>
    setToasts(prev => prev.filter(t => t.id !== id)), []);

  const show = useCallback((variant: ToastVariant, title: string, message?: string) => {
    const id = `${baseId}-${Date.now()}`;
    setToasts(prev => [...prev.slice(-3), { id, variant, title, message }]);
    if (variant !== 'error') setTimeout(() => dismiss(id), 4000); // error는 수동 닫기
  }, [baseId, dismiss]);

  return { toasts, show, dismiss };
}

// components/ui/BellButton.tsx — aria-label에 카운트 포함 (badge는 aria-hidden)
export function BellButton({ count, onClick }: { count: number; onClick: () => void }) {
  return (
    <button className="bell-btn" onClick={onClick}
      aria-label={count > 0 ? `알림 ${count}건 — 알림 목록 열기` : '알림 — 알림 목록 열기'}>
      🔔
      {count > 0 && <span className="bell-badge" aria-hidden="true">{count > 99 ? '99+' : count}</span>}
    </button>
  );
}

♿ 접근성 체크 (JM6)

  • error 알림: role="alert" + aria-live="assertive"
  • 나머지 알림: role="status" + aria-live="polite"
  • aria-atomic="true": 내용 전체를 한 번에 읽음
  • 벨 버튼: 카운트를 aria-label에 포함 — aria-hidden 뱃지는 보조
  • 닫기 버튼: 설명적 aria-label ("저장 완료 알림 닫기")
  • 토스트 컨테이너: position: fixed, z-index 최상위, 키보드로 접근 가능

// components/ui/OtpInput.tsx
'use client';
import { useRef } from 'react';

export function OtpInput({
  length = 6, value, onChange, disabled,
}: { length?: number; value: string[]; onChange: (v: string[]) => void; disabled?: boolean }) {
  const refs = useRef<(HTMLInputElement | null)[]>([]);
  const move = (i: number) => refs.current[i]?.focus();

  const handleChange = (i: number, char: string) => {
    const digit = char.replace(/\D/g, '').slice(-1);
    const next = [...value]; next[i] = digit; onChange(next);
    if (digit && i < length - 1) move(i + 1);
  };
  const handleKeyDown = (i: number, e: React.KeyboardEvent) => {
    if (e.key === 'Backspace'  && !value[i] && i > 0) move(i - 1);
    if (e.key === 'ArrowLeft'  && i > 0)              move(i - 1);
    if (e.key === 'ArrowRight' && i < length - 1)     move(i + 1);
  };
  const handlePaste = (e: React.ClipboardEvent) => {
    e.preventDefault();
    const digits = e.clipboardData.getData('text').replace(/\D/g, '').slice(0, length);
    const next = Array(length).fill('');
    digits.split('').forEach((d, i) => { next[i] = d; });
    onChange(next); move(Math.min(digits.length, length - 1));
  };

  return (
    <div role="group" aria-label={`${length}자리 인증 코드 입력`} className="otp-row">
      {Array.from({ length }, (_, i) => (
        <input key={i} ref={el => { refs.current[i] = el; }}
          className="otp-input" type="text" inputMode="numeric"
          pattern="[0-9]" maxLength={1} value={value[i] ?? ''} disabled={disabled}
          autoComplete={i === 0 ? 'one-time-code' : 'off'}
          aria-label={`인증 코드 ${i + 1}번째 자리`}
          onChange={e => handleChange(i, e.target.value)}
          onKeyDown={e => handleKeyDown(i, e)}
          onPaste={i === 0 ? handlePaste : undefined}
        />
      ))}
    </div>
  );
}
// ⚠ JM5: OTP 검증은 반드시 서버 API에서만 수행. 클라이언트 비교 절대 금지.

♿ 접근성 체크 (JM6)

  • OTP 그룹: role="group" + aria-label + 각 칸 개별 aria-label
  • 비밀번호 필드: <label> 필수 연결, aria-required="true"
  • 붙여넣기 지원: 첫 번째 칸에 onPaste 핸들러
  • inputmode="numeric": 모바일 숫자 키패드 자동 표시
  • 실패 메시지: aria-invalid="true" + aria-describedby
  • 재발송 버튼: 쿨다운 시간 aria-label에 포함
